cmake_minimum_required(VERSION 3.20)
project(r_anniversary VERSION 1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

include(CTest)
option(EYE_BUILD_APP "Build the raylib window (disable for CPU-only tests)" ON)
set(RAYLIB_ROOT "" CACHE PATH "Optional prebuilt raylib directory containing include/ and lib/")

add_library(ascii_scene STATIC scene.c celebration.c intro.c)
target_include_directories(ascii_scene PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
option(EYE_PARALLEL_RENDER "Use available OpenMP support for the dense character grid" ON)
if(EYE_PARALLEL_RENDER)
    find_package(OpenMP QUIET COMPONENTS C)
    if(OpenMP_C_FOUND)
        target_link_libraries(ascii_scene PUBLIC OpenMP::OpenMP_C)
    endif()
endif()
if(NOT MSVC)
    target_link_libraries(ascii_scene PUBLIC m)
endif()

function(eye_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()
eye_warnings(ascii_scene)

if(EYE_BUILD_APP)
    if(RAYLIB_ROOT)
        find_path(EYE_RAYLIB_INCLUDE raylib.h PATHS "${RAYLIB_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
        find_library(EYE_RAYLIB_LIBRARY NAMES raylib PATHS "${RAYLIB_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
        add_library(raylib UNKNOWN IMPORTED)
        set_target_properties(raylib PROPERTIES IMPORTED_LOCATION "${EYE_RAYLIB_LIBRARY}"
            INTERFACE_INCLUDE_DIRECTORIES "${EYE_RAYLIB_INCLUDE}")
        if(WIN32)
            set_property(TARGET raylib PROPERTY INTERFACE_LINK_LIBRARIES winmm opengl32 gdi32)
        endif()
    else()
        find_package(raylib 5.0 QUIET)
        if(NOT raylib_FOUND)
            include(FetchContent)
            set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
            set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
            FetchContent_Declare(raylib
                GIT_REPOSITORY https://github.com/raysan5/raylib.git
                GIT_TAG 5.5 GIT_SHALLOW TRUE)
            FetchContent_MakeAvailable(raylib)
        endif()
    endif()
    add_executable(r_anniversary main.c ascii_draw.c)
    target_link_libraries(r_anniversary PRIVATE ascii_scene raylib)
    eye_warnings(r_anniversary)
    if(BUILD_TESTING)
        add_test(NAME cli_help COMMAND r_anniversary --help)
        add_test(NAME cli_invalid_time COMMAND r_anniversary --time nan)
        add_test(NAME cli_invalid_resolution COMMAND r_anniversary --resolution 999999999999999999999x2160 --capture unused.png)
        add_test(NAME cli_resolution_requires_export COMMAND r_anniversary --resolution 3840x2160)
        set_tests_properties(cli_invalid_time cli_invalid_resolution cli_resolution_requires_export PROPERTIES WILL_FAIL TRUE)
    endif()
endif()

if(BUILD_TESTING)
    add_executable(scene_tests tests/scene_tests.c)
    target_link_libraries(scene_tests PRIVATE ascii_scene)
    eye_warnings(scene_tests)
    add_test(NAME scene_invariants COMMAND scene_tests)
endif()
